cmake_minimum_required(VERSION 3.21)
project(material_columns_host LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
foreach(name cursor_selection equipment_input native_inventory column_selection inventory_columns resource_routing slot_observation selection_state navigation runtime_adapter panel_binding outer_pool state_codec state_journal scroll_motion sort_panel memory_profile layout_resources diagnostic_log)
    add_executable(${name} ${name}.cpp)
    target_include_directories(${name} PRIVATE stubs)
    if(MSVC)
        target_compile_options(${name} PRIVATE /W4 /UNDEBUG)
        target_compile_definitions(${name} PRIVATE _CRT_SECURE_NO_WARNINGS)
    else()
        target_compile_options(${name} PRIVATE -Wall -Wextra -Werror -pedantic -UNDEBUG)
    endif()
    add_test(NAME ${name} COMMAND ${name})
endforeach()
