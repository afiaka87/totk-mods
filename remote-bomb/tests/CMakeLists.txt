cmake_minimum_required(VERSION 3.25)
project(remote_bomb_tests LANGUAGES CXX)
enable_testing()

function(remote_bomb_test name)
    add_executable(remote_bomb_${name}_tests ${ARGN})
    target_include_directories(remote_bomb_${name}_tests PRIVATE
        ../src/pure ../src/engine ../../runtime-support/include)
    target_compile_features(remote_bomb_${name}_tests PRIVATE cxx_std_20)
    if(MSVC)
        target_compile_options(remote_bomb_${name}_tests PRIVATE /W4 /WX /fp:strict)
    else()
        target_compile_options(remote_bomb_${name}_tests PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
    endif()
    add_test(NAME remote_bomb_${name} COMMAND remote_bomb_${name}_tests)
endfunction()
