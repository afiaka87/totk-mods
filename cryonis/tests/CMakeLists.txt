cmake_minimum_required(VERSION 3.25)
project(cryonis_tests LANGUAGES CXX)
enable_testing()
include_directories(../src/pure ../src/engine)
foreach(suite core presentation control)
    add_executable(cryonis_${suite}_tests ${suite}_tests.cpp)
    target_compile_features(cryonis_${suite}_tests PRIVATE cxx_std_20)
    if(MSVC)
        target_compile_options(cryonis_${suite}_tests PRIVATE /W4 /WX /fp:strict)
    else()
        target_compile_options(cryonis_${suite}_tests PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
    endif()
    add_test(NAME cryonis_${suite} COMMAND cryonis_${suite}_tests)
endforeach()
